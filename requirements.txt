# >= 1.2.2 clears a symlink-following flaw in set_key (arbitrary file overwrite
# via the cross-device rename fallback). This app only calls load_dotenv(), so
# it was never reachable here -- bumped to close the alert rather than to fix an
# exploitable path.
python-dotenv==1.2.3

# Required by st.login() / st.user (Streamlit's native OIDC). Streamlit needs
# Authlib>=1.3.2 but does not depend on it, so it has to be listed explicitly.
#
# DO NOT DOWNGRADE BELOW 1.6.12. Authlib is what verifies the Cognito ID token
# signature, and everything below that carries open advisories against exactly
# that job -- including a CRITICAL JWS/JWK header-injection signature-
# verification bypass and a fail-open OIDC hash binding flaw (both patched in
# 1.6.9), plus an open redirect patched in 1.6.12. A vulnerable Authlib means
# the login gate can be bypassed, not merely that a library is out of date.
#
# streamlit#13335 (Authlib 1.6.6 breaking st.login) is closed, fixed upstream
# in Streamlit, so it is no longer a reason to hold back. Re-run the gate tests
# in ecs-express/RUNBOOK-screening-tool.md 2.6 if you change this.
Authlib==1.8.0

# Required by Authlib, which does NOT declare it. Streamlit's st.login imports
# authlib.integrations.starlette_client, whose __init__ chain reaches
# authlib.integrations.httpx_client and does a bare `import httpx`. Authlib
# treats httpx as an optional extra, so nothing installs it and the login route
# dies with ModuleNotFoundError -- a 500 only at the moment a user clicks Log
# in, never at build or start-up. Authlib 1.3.2 lacked that import chain, so
# this became necessary when the pin moved to 1.8.0 for the CVE fixes.
httpx==0.28.1
